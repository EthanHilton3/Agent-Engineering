// Tile collision. Bodies are axis-aligned boxes { x, y, w, h, vx, vy } with x,y at the top-left.
var KB = (typeof KB !== 'undefined') ? KB : {};
if (typeof module !== 'undefined') Object.assign(KB, require('./constants.js'));

KB.Physics = (function () {
  const C = KB.C;
  const T = C.T;
  const EPS = 0.001;

  // The world: `tiles` is the live tile array (star blocks can disappear), `level` gives the size.
  function tileAt(world, tx, ty) {
    if (tx < 0 || tx >= world.w) return T.BRICK;  // invisible walls at both ends
    if (ty < 0 || ty >= world.h) return T.EMPTY;  // open sky above, pits below
    return world.tiles[ty * world.w + tx];
  }
  function setTile(world, tx, ty, t) {
    if (tx >= 0 && tx < world.w && ty >= 0 && ty < world.h) world.tiles[ty * world.w + tx] = t;
  }
  function isSolid(t) { return t === T.GROUND || t === T.BRICK || t === T.STAR_BLOCK; }
  function isSolidAt(world, tx, ty) { return isSolid(tileAt(world, tx, ty)); }

  function overlaps(a, b) {
    return a.x < b.x + b.w && a.x + a.w > b.x && a.y < b.y + b.h && a.y + a.h > b.y;
  }

  function stepX(body, world, sx) {
    const S = C.TILE;
    const nx = body.x + sx;
    const ty0 = Math.floor(body.y / S), ty1 = Math.floor((body.y + body.h - EPS) / S);
    const tx = sx > 0 ? Math.floor((nx + body.w - EPS) / S) : Math.floor(nx / S);
    for (let ty = ty0; ty <= ty1; ty++) {
      if (isSolidAt(world, tx, ty)) {
        body.x = sx > 0 ? tx * S - body.w : (tx + 1) * S;
        body.vx = 0;
        body.hitWall = sx > 0 ? 1 : -1;
        return true;
      }
    }
    body.x = nx;
    return false;
  }

  function stepY(body, world, sy) {
    const S = C.TILE;
    const ny = body.y + sy;
    const tx0 = Math.floor(body.x / S), tx1 = Math.floor((body.x + body.w - EPS) / S);
    const ty = sy > 0 ? Math.floor((ny + body.h - EPS) / S) : Math.floor(ny / S);
    for (let tx = tx0; tx <= tx1; tx++) {
      const t = tileAt(world, tx, ty);
      // Clouds only stop you from above: your feet must have been at or over the cloud top.
      const cloudStops = t === T.CLOUD && sy > 0 && body.y + body.h <= ty * S + EPS;
      if (isSolid(t) || cloudStops) {
        if (sy > 0) { body.y = ty * S - body.h; body.onGround = true; }
        else { body.y = (ty + 1) * S; body.hitCeiling = true; }
        body.vy = 0;
        return true;
      }
    }
    body.y = ny;
    return false;
  }

  // Move a body by (dx, dy) in sub-steps no larger than MAX_SUBSTEP so fast bodies can't skip tiles.
  function moveBody(body, world, dx, dy) {
    body.onGround = false;
    body.hitWall = 0;
    body.hitCeiling = false;
    const steps = Math.max(1, Math.ceil(Math.max(Math.abs(dx), Math.abs(dy)) / C.MAX_SUBSTEP));
    const sx = dx / steps, sy = dy / steps;
    let blockedX = false, blockedY = false;
    for (let i = 0; i < steps; i++) {
      if (!blockedX && sx !== 0) blockedX = stepX(body, world, sx);
      if (!blockedY && sy !== 0) blockedY = stepY(body, world, sy);
    }
    return body;
  }

  // True if a box overlaps any solid tile.
  function boxHitsSolid(world, x, y, w, h) {
    const S = C.TILE;
    for (let ty = Math.floor(y / S); ty <= Math.floor((y + h - EPS) / S); ty++) {
      for (let tx = Math.floor(x / S); tx <= Math.floor((x + w - EPS) / S); tx++) {
        if (isSolidAt(world, tx, ty)) return true;
      }
    }
    return false;
  }

  // Is there floor under the point (x, y)? Used by walkers to turn at ledges.
  function hasFloorAt(world, x, y) {
    const t = tileAt(world, Math.floor(x / C.TILE), Math.floor(y / C.TILE));
    return isSolid(t) || t === T.CLOUD;
  }

  // Walk the line between two points; true if no solid tile (other than `ignore` tiles) is in the way.
  function lineOfSight(world, x0, y0, x1, y1, ignore) {
    const S = C.TILE;
    const dist = Math.hypot(x1 - x0, y1 - y0);
    const steps = Math.max(1, Math.ceil(dist / (S / 4)));
    for (let i = 1; i < steps; i++) {
      const tx = Math.floor((x0 + (x1 - x0) * i / steps) / S);
      const ty = Math.floor((y0 + (y1 - y0) * i / steps) / S);
      if (ignore && ignore.tx === tx && ignore.ty === ty) continue;
      if (isSolidAt(world, tx, ty)) return false;
    }
    return true;
  }

  return { tileAt, setTile, isSolid, isSolidAt, overlaps, moveBody, boxHitsSolid, hasFloorAt, lineOfSight };
})();

  function updateEnemies(g) {
    const pc = center(g.player);
    const fallOut = g.world.h * C.TILE + C.TILE * 2;

      if (Math.abs(e.x + e.w / 2 - pc.x) > C.ACTIVATE_DIST) continue;

      else if (e.kind === 'flyer') updateFlyer(e);
      else if (e.kind === 'hopper') updateHopper(g, e, g.player);
      else if (e.kind === 'gordo') updateGordo(e);
      if (e.y > fallOut) e.removed = true;
    }

  }

    e.removed = true;
    const c = center(e);
    emit(g, 'defeat', { kind: e.kind, x: c.x, y: c.y });
    if (points) addScore(g, points, c.x, e.y);
  }

    const s = C.STAR_SIZE;
    return { id: nextId(g), kind: 'star', x: dir > 0 ? x - s / 3 : x - s + s / 3, y: y - s / 2, w: s, h: s,

    const s = C.PUFF_SIZE;
    return { id: nextId(g), kind: 'puff', x: dir > 0 ? x : x - s, y: y - s / 2, w: s, h: s,

  }

  function projectileHitsTiles(g, p) {
    const S = C.TILE;
    const tx = Math.floor((p.vx > 0 ? p.x + p.w : p.x) / S);
    for (let ty = Math.floor(p.y / S); ty <= Math.floor((p.y + p.h - 0.001) / S); ty++) {
      const t = P.tileAt(g.world, tx, ty);
      if (p.kind === 'star' && t === T.STAR_BLOCK) {
        P.setTile(g.world, tx, ty, T.EMPTY);
        emit(g, 'blockBreak', { x: tx * S + S / 2, y: ty * S + S / 2 });
      } else if (P.isSolid(t)) {
        emit(g, p.kind === 'star' ? 'starBurst' : 'puffGone', center(p));
        return false;
      }
    }
    return true;
  }

  function projectileHitsEnemies(g, p) {
    for (const e of g.enemies) {
      if (e.removed || e.harmless || p.hitIds.includes(e.id) || !P.overlaps(p, e)) continue;
      p.hitIds.push(e.id);
      if (e.kind === 'gordo') {           // nothing hurts Gordo
        emit(g, 'clink', center(p));
        return false;
      }
      if (p.kind === 'star') {            // Kirby's star pierces through a whole line of enemies
        defeatEnemy(g, e, C.SCORE_STAR_HIT * Math.pow(2, p.hits));
        p.hits++;
      } else {
        defeatEnemy(g, e, C.SCORE_PUFF_HIT);
        return false;
      }
    }
    return true;
  }

  function updateProjectiles(g) {
    for (const p of g.projectiles) {
      const steps = Math.max(1, Math.ceil(Math.abs(p.vx * C.DT) / C.MAX_SUBSTEP));
      let alive = true;
      for (let i = 0; i < steps && alive; i++) {
        p.x += p.vx * C.DT / steps;
        alive = projectileHitsTiles(g, p) && projectileHitsEnemies(g, p);
      }
      p.life--;
      if (alive && p.life <= 0) {
        emit(g, p.kind === 'star' ? 'starBurst' : 'puffGone', center(p));
        alive = false;
      }
      if (!alive) p.removed = true;
    }
    g.projectiles = g.projectiles.filter(p => !p.removed);
    g.enemies = g.enemies.filter(e => !e.removed);
  }

  return {
    emit, addScore, center,
    createEnemy, createBlock, createPickup, createStar, createPuff,
    updateEnemies, updateProjectiles, defeatEnemy,
  };
})();

KB.Inhale = (function () {

  const T = C.T;
  const P = KB.Physics;
  const E = KB.Entities;

  function mouth(p) {
    return { x: p.facing > 0 ? p.x + p.w : p.x, y: p.y + p.h * 0.55 };
  }

  // Is the point (x, y) inside the cone in front of Kirby's mouth?
  function pointInCone(p, x, y) {
    const m = mouth(p);
    const dx = (x - m.x) * p.facing;
    const dy = y - m.y;
    if (dx < -C.INHALE_BEHIND || dx > C.INHALE_RANGE) return false;
    return Math.abs(dy) <= C.INHALE_HALF_HEIGHT + Math.max(0, dx) * C.INHALE_SPREAD;
  }

  // Is this entity inside the cone with nothing solid in between? `tile` lets a star block ignore itself.
  function inCone(p, target, world, tile) {
    if (target.inhalable === false) return false;
    const c = E.center(target);
    if (!pointInCone(p, c.x, c.y)) return false;
    if (!world) return true;
    const m = mouth(p);
    return P.lineOfSight(world, m.x, m.y, c.x, c.y, tile);
  }

  // Star blocks inside the cone come loose and become pulled objects.
  function loosenBlocks(g, p) {
    const S = C.TILE;
    const m = mouth(p);
    const reach = C.INHALE_HALF_HEIGHT + C.INHALE_RANGE * C.INHALE_SPREAD;
    const x0 = Math.floor(Math.min(m.x, m.x + p.facing * C.INHALE_RANGE) / S);
    const x1 = Math.floor(Math.max(m.x, m.x + p.facing * C.INHALE_RANGE) / S);
    for (let ty = Math.floor((m.y - reach) / S); ty <= Math.floor((m.y + reach) / S); ty++) {
      for (let tx = x0; tx <= x1; tx++) {
        if (P.tileAt(g.world, tx, ty) !== T.STAR_BLOCK) continue;
        const box = { x: tx * S, y: ty * S, w: S, h: S };
        if (!inCone(p, box, g.world, { tx, ty })) continue;
        P.setTile(g.world, tx, ty, T.EMPTY);
        g.enemies.push(E.createBlock(g, tx, ty));
        E.emit(g, 'blockLoose', { x: tx * S + S / 2, y: ty * S + S / 2 });
      }
    }
  }

  function pullToward(o, m) {
    const c = E.center(o);
    const dx = m.x - c.x, dy = m.y - c.y;
    const dist = Math.hypot(dx, dy);
    if (dist < 0.001) return 0;
    const closeness = Math.max(0, 1 - dist / C.INHALE_RANGE);
    const step = Math.min(dist, (C.INHALE_PULL_SPEED + C.INHALE_PULL_BOOST * closeness) * C.DT);
    o.x += dx / dist * step;
    o.y += dy / dist * step;
    return dist - step;
  }

  // Let go of everything: enemies wake up again, loose blocks crumble.
  function releaseAll(g) {
    for (const e of g.enemies) {
      if (!e.pulled) continue;
      e.pulled = false;
      e.vx = 0; e.vy = 0;
      if (e.kind === 'block') {
        e.removed = true;
        E.emit(g, 'blockPop', E.center(e));
      }
    }
    for (const k of g.pickups) k.pulled = false;
    g.enemies = g.enemies.filter(e => !e.removed);
  }

  function update(g) {
    const p = g.player;
    if (p.mode !== 'inhale') {
      if (g.enemies.some(e => e.pulled) || g.pickups.some(k => k.pulled)) releaseAll(g);
      return;
    }
    // Once something is caught by the inhale it stays caught until Kirby lets go.
    for (const e of g.enemies) if (!e.pulled && inCone(p, e, g.world)) e.pulled = true;
    for (const k of g.pickups) if (!k.pulled && inCone(p, k, g.world)) k.pulled = true;
    loosenBlocks(g, p);

    const m = mouth(p);
    const swallowed = [];
    for (const e of g.enemies) {
      if (!e.pulled) continue;
      const left = pullToward(e, m);
      if (left <= C.INHALE_SWALLOW_DIST || P.overlaps(e, p)) {
        e.removed = true;
        swallowed.push(e.kind);
      }
    }
    for (const k of g.pickups) if (k.pulled) pullToward(k, m);
    g.enemies = g.enemies.filter(e => !e.removed);

    if (swallowed.length) {
      p.mode = 'full';
      p.held = { kinds: swallowed, count: swallowed.length };
      const enemies = swallowed.filter(k => k !== 'block').length;
      if (enemies) E.addScore(g, C.SCORE_SWALLOW * enemies, m.x, p.y);
      E.emit(g, 'swallow', { x: m.x, y: m.y, count: swallowed.length });
      releaseAll(g);
    }
  }

  function spit(g) {
    const p = g.player;
    const m = mouth(p);
    g.projectiles.push(E.createStar(g, m.x, m.y, p.facing));
    p.mode = 'normal';
    p.held = null;
    p.spitTimer = C.SPIT_ANIM_TICKS;
    E.emit(g, 'spit', { x: m.x, y: m.y });
  }

  function exhale(g) {
    const p = g.player;
    const m = mouth(p);
    g.projectiles.push(E.createPuff(g, m.x, m.y, p.facing));
    p.mode = 'normal';
    p.spitTimer = C.SPIT_ANIM_TICKS;
    E.emit(g, 'exhale', { x: m.x, y: m.y });
  }

  return { mouth, pointInCone, inCone, update, releaseAll, spit, exhale };
})();

  }

  function handleAction(g, p, inp) {
    if (inp.actionPressed) {
      if (p.mode === 'full') KB.Inhale.spit(g);
      else if (p.mode === 'fly') KB.Inhale.exhale(g);
      else if (p.mode === 'normal') {

    let accel = p.onGround ? C.GROUND_ACCEL : C.AIR_ACCEL;
    if (p.mode === 'fly') { max = C.FLY_SPEED_X; accel = C.FLY_ACCEL; }
    if (p.mode === 'full') max *= C.FULL_SPEED_FACTOR;
    if (p.mode === 'inhale') max *= C.INHALE_MOVE_FACTOR;
    const target = dir * max;
    const braking = target === 0 || Math.sign(target) !== Math.sign(p.vx);
    const rate = (p.onGround && braking ? C.GROUND_DECEL : accel) * C.DT;
    p.vx = approach(p.vx, target, rate);
  }

  function handleJump(g, p, inp) {
    if (inp.jumpPressed) p.jumpBuffer = C.JUMP_BUFFER_TICKS;
    const airborne = !p.onGround && p.coyote <= 0;
    const canFly = p.mode === 'normal' || p.mode === 'fly';

    if (inp.jumpPressed && airborne && canFly) {

      } else {
        E.emit(g, 'tired', { x: p.x + p.w / 2, y: p.y });
      }
      p.jumping = false;
      p.jumpBuffer = 0;
    } else if (p.jumpBuffer > 0 && canJump) {
      p.vy = -C.JUMP_SPEED;
      p.jumping = true;
      p.jumpBuffer = 0;
      p.coyote = 0;

    if (p.mode === 'fly') {
      p.vy = Math.min(p.vy + C.FLY_GRAVITY * C.DT, C.FLY_MAX_FALL);
    } else {
      p.vy = Math.min(p.vy + (p.vy < 0 ? C.GRAVITY : C.FALL_GRAVITY) * C.DT, C.MAX_FALL_SPEED);
    }
  }

  function update(g, inp) {
    const p = g.player;
    p.anim++;
    if (p.invuln > 0) p.invuln--;
    if (p.spitTimer > 0) p.spitTimer--;
    if (p.landTimer > 0) p.landTimer--;
    if (p.jumpBuffer > 0) p.jumpBuffer--;
    if (p.coyote > 0) p.coyote--;

    handleAction(g, p, inp);
    handleRun(p, inp);
    handleJump(g, p, inp);
    applyGravity(p);

      p.flaps = 0;
      p.jumping = false;

        E.emit(g, 'land', { x: p.x + p.w / 2, y: p.y + p.h, speed: fallSpeed });
      }
    }
  }

  return { create, update };
})();

// Keyboard -> input snapshots. The key handling is plain functions so tests can feed it fake events.
var KB = (typeof KB !== 'undefined') ? KB : {};

KB.Input = (function () {
  // Only the keys the game asks for. Enter is matched on `key`, so the keypad Enter works too.
  const BY_CODE = {
    ArrowLeft: 'left', KeyA: 'left',
    ArrowRight: 'right', KeyD: 'right',
    Space: 'jump',
  };
  const BY_KEY = { Enter: 'action' };
  const UTILITY = { KeyP: 'pause', KeyM: 'mute', KeyR: 'restart' };

  function create() {
    return {
      held: { left: false, right: false, jump: false, action: false },
      pressed: { jump: 0, action: 0, pause: 0, mute: 0, restart: 0 },
    };
  }

  function actionFor(ev) { return BY_KEY[ev.key] || BY_CODE[ev.code] || null; }

  // Feed one keydown/keyup event. Returns true if the game used it.
  function handleEvent(state, ev) {
    const down = ev.type === 'keydown';
    const action = actionFor(ev);
    if (action) {
      if (down && !ev.repeat && !state.held[action] && action in state.pressed) state.pressed[action]++;
      state.held[action] = down;
    } else if (UTILITY[ev.code]) {
      if (down && !ev.repeat) state.pressed[UTILITY[ev.code]]++;
    } else {
      return false;
    }
    if (ev.preventDefault) ev.preventDefault();
    return true;
  }

  function releaseAll(state) {
    Object.keys(state.held).forEach(k => { state.held[k] = false; });
  }

  // What the game sees for one tick. Presses are latched until the next snapshot, so a quick tap
  // between two ticks is never lost.
  function snapshot(state) {
    const h = state.held, p = state.pressed;
    const snap = {
      left: h.left, right: h.right, jump: h.jump, action: h.action,
      jumpPressed: p.jump > 0, actionPressed: p.action > 0,
      pausePressed: p.pause > 0, mutePressed: p.mute > 0, restartPressed: p.restart > 0,
    };
    Object.keys(p).forEach(k => { p[k] = 0; });
    return snap;
  }

  function attach(target, state, onKey) {
    const handler = ev => { if (handleEvent(state, ev) && onKey) onKey(ev); };
    target.addEventListener('keydown', handler);
    target.addEventListener('keyup', handler);
    target.addEventListener('blur', () => releaseAll(state));
  }

  return { create, handleEvent, snapshot, releaseAll, attach };
})();

function connect(url) {
  let nextId = 0;
  const pending = new Map();
  const listeners = new Map();
  ws = new WebSocket(url);
  ws.onmessage = ev => {
    const msg = JSON.parse(ev.data);
    if (msg.id && pending.has(msg.id)) {
      const { resolve, reject, method } = pending.get(msg.id);
      pending.delete(msg.id);
      if (msg.error) reject(new Error(method + ': ' + msg.error.message)); else resolve(msg.result);
    } else if (msg.method && listeners.has(msg.method)) {
      listeners.get(msg.method).forEach(fn => fn(msg.params));
    }
  };
  const send = (method, params = {}) => new Promise((resolve, reject) => {
    const id = ++nextId;
    pending.set(id, { resolve, reject, method });
    ws.send(JSON.stringify({ id, method, params }));
  });
  const on = (method, fn) => { if (!listeners.has(method)) listeners.set(method, []); listeners.get(method).push(fn); };
  return new Promise((resolve, reject) => {
    ws.onopen = () => resolve({ send, on });
    ws.onerror = () => reject(new Error('WebSocket failed'));

// A scripted player for the playthrough test. It looks at the game state (read only) and answers
// with simulated keyboard events fed through input.js. It never moves Kirby or edits state directly.
const { C } = require('../js/logic/constants.js');
const { Physics } = require('../js/logic/physics.js');
const { Inhale } = require('../js/logic/inhale.js');
const { Input } = require('../js/input.js');
const { Game } = require('../js/logic/game.js');

const S = C.TILE;
const KEY = {
  right: { code: 'ArrowRight', key: 'ArrowRight' },
  jump: { code: 'Space', key: ' ' },
  action: { code: 'Enter', key: 'Enter' },
};

function createBot(game) {
  const input = Input.create();
  const down = { right: false, jump: false, action: false };
  const bot = { state: 'walk', timer: 0, targetY: 0, log: [] };

  function set(name, isDown) {
    if (down[name] === isDown) return;
    down[name] = isDown;
    Input.handleEvent(input, Object.assign({ type: isDown ? 'keydown' : 'keyup', repeat: false }, KEY[name]));
  }
  // A tap is a keydown now and a keyup on the next tick (only if the key was up).
  function tap(name) { if (!down[name]) set(name, true); }

  const g = () => game;
  const tile = (tx, ty) => Physics.tileAt(g().world, tx, ty);
  const pitColumn = tx => tile(tx, g().world.h - 1) === C.T.EMPTY;

  function enemyAhead(maxDist, inhalableOnly) {
    const p = g().player;
    let best = null;
    for (const e of g().enemies) {
      if (e.kind === 'block' || (inhalableOnly && !e.inhalable)) continue;
      const dx = e.x - (p.x + p.w);
      const dy = (e.y + e.h / 2) - (p.y + p.h / 2);
      if (dx > -e.w && dx < maxDist && Math.abs(dy) < S * 1.5 && (!best || dx < best.dx)) best = { e, dx, dy };
    }
    return best;
  }

  function starBlockAhead() {
    const p = g().player;
    const tx = Math.floor((p.x + p.w + S * 1.5) / S);
    const rows = [Math.floor(p.y / S), Math.floor((p.y + p.h - 1) / S)];
    return rows.some(ty => tile(tx, ty) === C.T.STAR_BLOCK || tile(tx - 1, ty) === C.T.STAR_BLOCK);
  }

  // Height of the obstacle right in front of Kirby (0 if none): a wall in his way.
  function wallTopAhead() {
    const p = g().player;
    const tx = Math.floor((p.x + p.w + 3) / S);
    const ty1 = Math.floor((p.y + p.h - 1) / S);
    if (!Physics.isSolidAt(g().world, tx, ty1) || tile(tx, ty1) === C.T.STAR_BLOCK) return 0;
    let ty = ty1;
    while (ty > 0 && Physics.isSolidAt(g().world, tx, ty - 1)) ty--;
    return ty * S;
  }

  function gapAhead() {
    const p = g().player;
    const front = Math.floor((p.x + p.w + 6) / S);
    return pitColumn(front);
  }

  function overSafeGround() {
    const p = g().player;
    return !pitColumn(Math.floor(p.x / S)) && !pitColumn(Math.floor((p.x + p.w) / S)) && !gapAhead();
  }

  function startCross(targetY, why) {
    bot.state = 'cross';
    bot.targetY = targetY;
    bot.timer = 0;
    bot.log.push(why + ' at x=' + Math.round(g().player.x));
  }

  // Decide this tick's keys. Called once per tick before the game updates.
  function think() {
    const game = g();
    const p = game.player;
    // Release taps from the previous tick so the next press is a fresh one.
    if (down.jump && bot.state !== 'hold-jump') set('jump', false);
    if (game.mode === 'title') { set('action', false); tap('action'); return; }
    if (game.mode !== 'playing') { set('right', false); set('action', false); bot.state = 'walk'; return; }
    bot.timer++;

    if (p.mode === 'full') {          // spit as soon as something is in our mouth
      set('right', false);
      if (down.action) set('action', false); else tap('action');
      bot.state = 'walk';
      return;
    }

    if (bot.state === 'inhale') {
      set('right', false);
      const target = enemyAhead(C.INHALE_RANGE, true);
      if (bot.timer > 90 || (!target && !starBlockAhead() && !game.enemies.some(e => e.pulled))) {
        set('action', false);
        bot.state = 'walk';
      } else set('action', true);
      return;
    }
    if (down.action) set('action', false);

    if (bot.state === 'cross') {
      const high = p.y <= bot.targetY + 4;
      set('right', high || !p.onGround && bot.climbed);
      if (high) bot.climbed = true;
      // Puff away anything in the flight path.
      const threat = enemyAhead(36, true);
      if (p.mode === 'fly' && threat && Math.abs(threat.dy) < 14) { tap('action'); return; }
      const wantsAltitude = !overSafeGround() || p.y > bot.targetY + 4 && !bot.climbed;
      if (p.onGround && !bot.climbed) tap('jump');
      else if (wantsAltitude && p.vy > 0 && p.y > bot.targetY) tap('jump');
      else if (!p.onGround && p.mode !== 'fly' && !overSafeGround()) tap('jump');
      if (p.onGround && bot.climbed && overSafeGround()) { bot.state = 'walk'; bot.climbed = false; }
      if (bot.timer > 60 * 20) { bot.state = 'walk'; bot.climbed = false; }
      return;
    }

    // Walking.
    const target = enemyAhead(C.INHALE_RANGE - 4, true);
    if (p.onGround && target && Inhale.inCone(p, target.e, game.world)) {
      set('right', false); set('action', true);
      bot.state = 'inhale'; bot.timer = 0;
      return;
    }
    if (p.onGround && starBlockAhead()) {
      set('right', false); set('action', true);
      bot.state = 'inhale'; bot.timer = 0;
      return;
    }
    // Hoppers in mid-air or Gordo close by: wait for a better moment.
    const wary = enemyAhead(S * 3, false);
    if (wary && wary.e.kind === 'hopper' && !wary.e.onGround) { set('right', false); return; }
    if (p.onGround && gapAhead()) { set('right', false); startCross(S * 2.5, 'gap'); return; }
    const wall = wallTopAhead();
    if (p.onGround && wall) { set('right', false); startCross(wall - p.h - S, 'wall'); return; }
    set('right', true);
  }

  return { input, think, bot, down };
}

// Run a whole game with the bot. Returns a summary for the test to check.
function playThrough(maxTicks) {
  const game = Game.create();
  const { input, think, bot } = createBot(game);
  const { Level } = require('../js/level.js');
  const gaps = Level.findGaps(game.level).map(gp => ({ ...gp, crossed: false }));
  const startLives = game.lives;
  let ticks = 0;
  let deaths = 0;
  for (; ticks < maxTicks && game.mode !== 'won' && game.mode !== 'gameover'; ticks++) {
    think();
    Game.update(game, Input.snapshot(input));
    if (game.events.some(e => e.type === 'die')) deaths++;
    const p = game.player;
    if (game.mode === 'playing' && p.onGround) {
      for (const gp of gaps) if (p.x > (gp.to + 1) * S) gp.crossed = true;
    }
  }
  return { game, ticks, gaps, deaths, startLives, log: bot.log };
}

I've built the Kirby platformer and it works. All 49 unit tests pass, the browser smoke test passes, and the automated playthrough finishes the level using only simulated key presses. It crosses all 5 gaps without losing a life, and it has to inhale and spit to get through the star-block wall.

How to run: double-click index.html. There's no build step, nothing to install and no external files.

How to run the tests (from the project folder):

node --test runs the unit tests and the playthrough.
node tests/smoke.mjs opens the game in headless Chrome, plays it with real key events, fails on any console error, and saves screenshots to .tmp/. Set CHROME_PATH if Chrome isn't in the usual place. It shuts Chrome down and deletes its temporary profile afterwards; I confirmed no Chrome processes were left running.
Checked against your request:

Kirby: runs, jumps, and flies by tapping Space in the air. Holding Enter pulls in enemies, star blocks and pickups in front of him. Once he's swallowed something, Enter spits it out as a star that defeats enemies.
Level: one side-scrolling level with platforms, clouds you can jump up through, 5 gaps, 2 checkpoints, and a Warp Star at the end as the goal.
Enemies: four kinds. A walker, a flyer that bobs up and down, a hopper that jumps toward Kirby, and Gordo, a spiky one that can't be inhaled or defeated.
Lives and screens: lives, score, a game-over screen, a win screen, and restart (Enter or Space on the end screens, or R at any time).
Controls: the bindings are exactly what you listed, and they're shown on screen in three places: a strip along the bottom, the title screen, and a line under the game.
Art: every sprite, tile and letter is drawn in code. Pixels stay sharp because the game only scales up in whole steps. There are animations for idle, blink, run, jump, fall, fly, inhale, mouthful, spit, landing and hurt.
Where I made a choice or approximated:

Flying is limited to 8 flaps before Kirby has to land, with a meter in the top bar. He also can't fly above a ceiling just under that bar. The real games allow endless flight.
Mouthful Kirby can't fly, as in the original games, so you have to spit before crossing a gap.
"Surprisingly strong" is shown through his abilities rather than a stat: he pulls star blocks straight out of walls, and one spat star goes through a whole line of enemies.
The palette is one fixed set of about 45 named colours. It's limited, but not as strict as a real retro console.
Extras I added: health (3 hits per life) and a tomato that refills it, collectible stars, checkpoints, the hopper and Gordo, an air puff when you press Enter while flying or when you land from flying, slower falling while inhaling in mid-air, a wall of star blocks you have to inhale through, double points for each extra enemy one star hits, enemies being knocked out when they bump into Kirby, a saved high score, a title screen, tutorial signs, pause (P), mute (M), sound effects and a music loop, particles and screen shake, and small movement aids (shorter hops when you let go of Space early, and a brief window to jump after walking off a ledge).

What I couldn't verify:

Sound: I couldn't hear anything. The test runs without errors with sound switched on, but whether it sounds good is untested.
How it feels to play: I set the jump, fly and inhale numbers by calculation and the scripted run; I haven't played it by hand. All of them are in one block in js/logic/constants.js if you want to tune them.
Known problems:

On screens faster than 60Hz, movement may look slightly less smooth, because the game updates 60 times a second and doesn't smooth between updates.
Kirby's right cheek blush touches his outline when he faces right. It's cosmetic only.
Defeated enemies stay gone after you respawn at a checkpoint. That's deliberate, but it does make retries easier.
The playthrough proves the level can be finished, but the scripted player still took one hit along the way.
